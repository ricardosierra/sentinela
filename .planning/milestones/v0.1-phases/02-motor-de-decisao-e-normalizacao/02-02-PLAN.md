```kotlin
package org.sentinela.app.domain

class CallDecisionEngine {
    fun decide(call: ScreenedCall, settings: ScreeningSettings,
               contact: ContactLookup, whitelist: WhitelistLookup): CallDecision
}

data class ScreenedCall(val direction: CallDirection, val number: ScreenedNumber)
enum class CallDirection { INCOMING, OUTGOING }
sealed interface ScreenedNumber {
    data class Valid(val e164: String) : ScreenedNumber
    data object Private : ScreenedNumber
    data object Invalid : ScreenedNumber
}
enum class ContactLookup { HIT, MISS, UNAVAILABLE }
enum class WhitelistLookup { HIT, MISS, LOOKUP_FAILED }

sealed interface CallDecision { val reason: DecisionReason }
// Allow, Silence, Reject, SendSilentlyToVoicemail, BlockWithoutTrace — todos data class(reason)

enum class DecisionReason(val code: String) {
    OUTGOING_CALL, PROTECTION_DISABLED, PRIVATE_NUMBER, CONTACT, PERSONAL_WHITELIST,
    UNKNOWN_NUMBER, INVALID_NUMBER, LOCAL_LOOKUP_FAILURE, FALLBACK_POLICY
}
```
```kotlin
package org.sentinela.app.settings
data class ScreeningSettings(
    val protectionEnabled: Boolean = true,
    val unknownPolicy: OriginPolicy = OriginPolicy.BLOCK,
    val contactsPolicy: OriginPolicy = OriginPolicy.RING,
    val whitelistPolicy: OriginPolicy = OriginPolicy.NEVER_SILENCE,
    val blockPrivateNumbers: Boolean = true,
    val blockMode: BlockMode = BlockMode.REJECT,
    val hideFromNativeCallLog: Boolean = true,
    val showOwnNotification: Boolean = false,
    val fallbackPolicy: FallbackPolicy = FallbackPolicy.ALLOW,
)
enum class OriginPolicy { RING, BLOCK, SILENCE, NEVER_SILENCE }
enum class BlockMode { REJECT, SILENT_VOICEMAIL }
enum class FallbackPolicy { ALLOW, BLOCK }
```
