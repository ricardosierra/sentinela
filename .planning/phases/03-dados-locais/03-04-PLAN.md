```kotlin
package org.sentinela.app.data.local

import kotlinx.coroutines.CoroutineDispatcher
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map
import kotlinx.coroutines.withContext
import org.sentinela.app.data.local.db.WhitelistDao
import org.sentinela.app.data.local.db.WhitelistEntity

/**
 * Implementacao Room da whitelist pessoal.
 *
 * A chave persistida e a saida do PhoneNumberNormalizer (contrato da Fase 2):
 * E.164 para numeros normais, digitos crus abaixo de PhoneNumbers.LIMIAR_CURTO (6).
 * A normalizacao acontece ANTES de chegar aqui — este repositorio nao normaliza.
 */
class RoomWhitelistRepository(
    private val dao: WhitelistDao,
    private val io: CoroutineDispatcher = Dispatchers.IO,
) : PersonalWhitelistRepository {

    /**
     * Caminho quente: delega ao DAO NAO-suspend, evitando o hop interno de
     * dispatcher do Room (medido: p95 9,12 ms suspend vs 3,59 ms nao-suspend).
     */
    override suspend fun contains(numberE164: String): Boolean =
        withContext(io) { dao.containsBlocking(numberE164) }

    override fun observeAll(): Flow<List<WhitelistEntry>> =
        dao.observeAll().map { list -> list.map(WhitelistEntity::toDomain) }

    fun search(query: String): Flow<List<WhitelistEntry>> =
        dao.search(query).map { list -> list.map(WhitelistEntity::toDomain) }

    /**
     * Dedup (WLT-04): o indice UNICO em number_key garante a atomicidade, mas
     * @Upsert com id = 0 nao casa pelo indice — ele faz UPDATE por chave primaria
     * e nao encontraria linha. Por isso o id existente e resolvido antes.
     * Duplicata vira atualizacao silenciosa: sem erro visivel ao usuario.
     */
    override suspend fun upsert(entry: WhitelistEntry) = withContext(io) {
        val existing = dao.findByKey(entry.numberE164)
        dao.upsert(
            entry.toEntity(
                id = existing?.id ?: entry.id,
                // createdAt original preservado: reeditar nao "rejuvenesce" a entrada.
                createdAtUtcMillis = existing?.createdAtUtcMillis ?: entry.createdAtUtcMillis,
            ),
        )
    }

    override suspend fun delete(id: Long) = withContext(io) { dao.deleteById(id) }
}
```
Mais as funcoes de mapeamento `WhitelistEntity.toDomain()` e
`WhitelistEntry.toEntity(id, createdAtUtcMillis)` no mesmo arquivo (privadas/internas).

```kotlin
private const val TEST_DB = "perf-test.db"
private const val ENTRIES = 1_000
private const val WARMUP = 300    // medido: sem warmup a cauda dobra

@Before fun setUp() {
    val ctx = ApplicationProvider.getApplicationContext<Context>()

    db.openHelper.writableDatabase.run {

                        "VALUES ('+55119${"%08d".format(i)}', NULL, 1, 0)",
                )
            }
            setTransactionSuccessful()
        } finally { endTransaction() }
    }
}

@After fun tearDown() {
    db.close()
    ApplicationProvider.getApplicationContext<Context>().deleteDatabase(TEST_DB)
}
```

```kotlin
@Test fun containsUsaIndiceEmVezDeFullScan() {
    val sql = "SELECT EXISTS(SELECT 1 FROM whitelist WHERE number_key = ? AND enabled = 1)"
    val plan = buildString {
        db.openHelper.writableDatabase
            .query("EXPLAIN QUERY PLAN $sql", arrayOf("+5511900000500"))
            .use { c -> while (c.moveToNext()) appendLine(c.getString(c.columnCount - 1)) }

    println("SENTINELA|EQP|$plan")
    assertTrue("plano sem indice:\n$plan", plan.contains("USING INDEX index_whitelist_number_key"))
    assertFalse("full scan detectado:\n$plan", plan.contains("SCAN whitelist"))
}
```
